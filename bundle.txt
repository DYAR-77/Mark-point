package com.markpoint;

import net.minecraft.network.PacketByteBuf;

import java.util.UUID;

public record MarkData(UUID id, UUID owner, String ownerName, int type,
					   double x, double y, double z,
					   String label, int color, int durationSec) {

	public static final int MAX_LABEL = 32;

	public void write(PacketByteBuf buf) {
		buf.writeUuid(id);
		buf.writeUuid(owner);
		buf.writeString(ownerName == null ? "" : ownerName, 64);
		buf.writeInt(type);
		buf.writeDouble(x);
		buf.writeDouble(y);
		buf.writeDouble(z);
		buf.writeString(label == null ? "" : label, 64);
		buf.writeInt(color);
		buf.writeInt(durationSec);
	}

	public static MarkData read(PacketByteBuf buf) {
		return new MarkData(
				buf.readUuid(),
				buf.readUuid(),
				buf.readString(64),
				buf.readInt(),
				buf.readDouble(),
				buf.readDouble(),
				buf.readDouble(),
				buf.readString(64),
				buf.readInt(),
				buf.readInt());
	}

	public MarkData withDuration(int seconds) {
		return new MarkData(id, owner, ownerName, type, x, y, z, label, color, seconds);
	}
}

/** Server-side permissions for sharing marks. File: config/markpoint-server.properties */
public final class ServerConfig {
	public boolean shareEnabled = true;
	public int maxMarksPerPlayer = 10;
	public int maxDurationSec = 600;
	public boolean allowPermanent = true;
	public int maxRange = 4096;

	private static Path path() {
		return FabricLoader.getInstance().getConfigDir().resolve("markpoint-server.properties");
	}

	public static ServerConfig load() {
		ServerConfig c = new ServerConfig();
		Path p = path();
		if (Files.exists(p)) {
			Properties props = new Properties();
			try (Reader r = Files.newBufferedReader(p)) {
				props.load(r);
				c.shareEnabled = Boolean.parseBoolean(props.getProperty("shareEnabled", "true"));
				c.maxMarksPerPlayer = Math.max(1, parse(props, "maxMarksPerPlayer", 10));
				c.maxDurationSec = Math.max(0, parse(props, "maxDurationSec", 600));
				c.allowPermanent = Boolean.parseBoolean(props.getProperty("allowPermanent", "true"));
				c.maxRange = Math.max(16, parse(props, "maxRange", 4096));
			} catch (IOException ignored) {
			}
		} else {
			c.save();
		}
		return c;
	}

	private static int parse(Properties props, String key, int def) {
		try {
			return Integer.parseInt(props.getProperty(key, String.valueOf(def)).trim());
		} catch (NumberFormatException e) {
			return def;
		}
	}

	public void save() {
		Properties props = new Properties();
		props.setProperty("shareEnabled", String.valueOf(shareEnabled));
		props.setProperty("maxMarksPerPlayer", String.valueOf(maxMarksPerPlayer));
		props.setProperty("maxDurationSec", String.valueOf(maxDurationSec));
		props.setProperty("allowPermanent", String.valueOf(allowPermanent));
		props.setProperty("maxRange", String.valueOf(maxRange));
		try (Writer w = Files.newBufferedWriter(path())) {
			props.store(w, "MarkPoint server config (maxDurationSec 0 = unlimited)");
		} catch (IOException ignored) {
		}
	}

package com.markpoint.server;

import com.markpoint.MarkData;
import com.markpoint.MarkType;
import com.markpoint.network.MarkPayload;
import com.markpoint.network.RemovePayload;
import net.fabricmc.fabric.api.event.lifecycle.v1.ServerLifecycleEvents;
import net.fabricmc.fabric.api.event.lifecycle.v1.ServerTickEvents;
import net.fabricmc.fabric.api.networking.v1.ServerPlayConnectionEvents;
import net.fabricmc.fabric.api.networking.v1.ServerPlayNetworking;
import net.minecraft.network.packet.CustomPayload;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;

import java.util.Iterator;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.UUID;

public final class ServerMarkManager {
	private record Entry(MarkData data, long expiresAt) {
	}

	private static final Map<UUID, Entry> MARKS = new LinkedHashMap<>();
	private static ServerConfig cfg = new ServerConfig();

	private ServerMarkManager() {
	}

	public static void init() {
		ServerLifecycleEvents.SERVER_STARTING.register(server -> {
			MARKS.clear();
			cfg = ServerConfig.load();
		});
		ServerLifecycleEvents.SERVER_STOPPED.register(server -> MARKS.clear());

		ServerPlayNetworking.registerGlobalReceiver(MarkPayload.ID,
				(payload, ctx) -> onCreate(ctx.player(), ctx.server(), payload.data()));
		ServerPlayNetworking.registerGlobalReceiver(RemovePayload.ID,
				(payload, ctx) -> onRemove(ctx.player(), ctx.server(), payload.id()));

		ServerPlayConnectionEvents.JOIN.register((handler, sender, server) -> {
			if (!cfg.shareEnabled) return;
			long now = System.currentTimeMillis();
			for (Entry e : MARKS.values()) {
				int dur = 0;
				if (e.expiresAt() > 0) {
					dur = (int) Math.max(1, (e.expiresAt() - now + 999) / 1000);
				}
				sender.sendPacket(new MarkPayload(e.data().withDuration(dur)));
			}
		});

		ServerTickEvents.END_SERVER_TICK.register(server -> {
			if (server.getTicks() % 20 != 0 || MARKS.isEmpty()) return;
			long now = System.currentTimeMillis();
			Iterator<Entry> it = MARKS.values().iterator();
			while (it.hasNext()) {
				Entry e = it.next();
				if (e.expiresAt() > 0 && now >= e.expiresAt()) {
					it.remove();
					broadcast(server, new RemovePayload(e.data().id()));
				}
			}
		});
	}

	private static void onCreate(ServerPlayerEntity p, MinecraftServer server, MarkData d) {
		if (!cfg.shareEnabled) return;

		double dx = d.x() - p.getX();
		double dy = d.y() - p.getY();
		double dz = d.z() - p.getZ();
		double distSq = dx * dx + dy * dy + dz * dz;
		if (!Double.isFinite(distSq) || distSq > (double) cfg.maxRange * cfg.maxRange) return;

		int dur = Math.max(0, d.durationSec());
		if (dur == 0 && !cfg.allowPermanent) {
			dur = cfg.maxDurationSec > 0 ? Math.min(60, cfg.maxDurationSec) : 60;
		} else if (dur > 0 && cfg.maxDurationSec > 0 && dur > cfg.maxDurationSec) {
			dur = cfg.maxDurationSec;
		}

		String label = d.label() == null ? "" : d.label();
		if (label.length() > MarkData.MAX_LABEL) label = label.substring(0, MarkData.MAX_LABEL);

		int type = Math.floorMod(d.type(), MarkType.values().length);
		UUID owner = p.getUuid();

		// enforce per-player limit: drop the oldest marks of this player
		while (countOf(owner) >= cfg.maxMarksPerPlayer) {
			UUID oldest = null;
			for (Entry e : MARKS.values()) {
				if (e.data().owner().equals(owner)) {
					oldest = e.data().id();
					break;
				}
			}
			if (oldest == null) break;
			MARKS.remove(oldest);
			broadcast(server, new RemovePayload(oldest));
		}

		MarkData clean = new MarkData(UUID.randomUUID(), owner, p.getName().getString(), type,
				d.x(), d.y(), d.z(), label, d.color() | 0xFF000000, dur);
		long expires = dur > 0 ? System.currentTimeMillis() + dur * 1000L : 0;
		MARKS.put(clean.id(), new Entry(clean, expires));
		broadcast(server, new MarkPayload(clean));
	}

	private static void onRemove(ServerPlayerEntity p, MinecraftServer server, UUID id) {
		Entry e = MARKS.get(id);
		if (e == null || !e.data().owner().equals(p.getUuid())) return;
		MARKS.remove(id);
		broadcast(server, new RemovePayload(id));
	}

	private static int countOf(UUID owner) {
		int n = 0;
		for (Entry e : MARKS.values()) {
			if (e.data().owner().equals(owner)) n++;
		}
		return n;
	}

	private static void broadcast(MinecraftServer server, CustomPayload payload) {
		for (ServerPlayerEntity pl : server.getPlayerManager().getPlayerList()) {
			if (ServerPlayNetworking.canSend(pl, payload.getId())) {
				ServerPlayNetworking.send(pl, payload);
			}
		}
	}
}

import net.minecraft.util.hit.HitResult;
import net.minecraft.util.math.Vec3d;

import java.util.ArrayList;
import java.util.Collection;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

public final class ClientMarks {
	public static final class Mark {
		public final MarkData data;
		public final long expiresAt;

		Mark(MarkData data, long expiresAt) {
			this.data = data;
			this.expiresAt = expiresAt;
		}
	}

	private static final Map<UUID, Mark> MARKS = new LinkedHashMap<>();

	private ClientMarks() {
	}

	public static Collection<Mark> all() {
		return MARKS.values();
	}

	public static void add(MarkData d) {
		long exp = d.durationSec() > 0 ? System.currentTimeMillis() + d.durationSec() * 1000L : 0;
		MARKS.put(d.id(), new Mark(d, exp));
		playSound(MarkType.byOrdinal(d.type()));
	}

	public static void remove(UUID id) {
		MARKS.remove(id);
	}

	public static void clear() {
		MARKS.clear();
	}

	public static void prune() {
		long now = System.currentTimeMillis();
		MARKS.values().removeIf(m -> m.expiresAt > 0 && now >= m.expiresAt);
	}

	private static void playSound(MarkType type) {
		if (!ClientConfig.sounds) return;
		float pitch = (type == MarkType.ENEMY || type == MarkType.DANGER) ? 0.7f : 1.3f;
		MinecraftClient.getInstance().getSoundManager().play(
				PositionedSoundInstance.master(SoundEvents.BLOCK_NOTE_BLOCK_BELL.value(), pitch));
	}

	/** Point the player is looking at (entity, block, or far point in the air). */
	public static Vec3d aimPoint(MinecraftClient mc) {
		if (mc.player == null) return Vec3d.ZERO;
		HitResult hit = mc.player.raycast(256.0, 1.0f, false);
		if (mc.crosshairTarget instanceof EntityHitResult e) {
			return e.getPos();
		}
		if (hit instanceof BlockHitResult b && hit.getType() == HitResult.Type.BLOCK) {
			return Vec3d.ofCenter(b.getBlockPos()).add(0, 1.0, 0);
		}
		return hit.getPos();
	}

	public static void create(MarkType type, Vec3d pos) {
		MinecraftClient mc = MinecraftClient.getInstance();
		if (mc.player == null) return;
		String label = type == MarkType.CUSTOM ? ClientConfig.customLabel : "";
		if (label.length() > MarkData.MAX_LABEL) label = label.substring(0, MarkData.MAX_LABEL);
		MarkData d = new MarkData(UUID.randomUUID(), mc.player.getUuid(), mc.player.getName().getString(),
				type.ordinal(), pos.x, pos.y, pos.z, label,
				ClientConfig.colors[type.ordinal()], ClientConfig.duration);
		if (ClientPlayNetworking.canSend(MarkPayload.ID)) {
			ClientPlayNetworking.send(new MarkPayload(d)); // server echoes it back to everyone
		} else {
			add(d); // local-only mark
		}
	}

	public static void clearMine() {
		MinecraftClient mc = MinecraftClient.getInstance();
		if (mc.player == null) return;
		UUID me = mc.player.getUuid();
		List<UUID> ids = new ArrayList<>();
		for (Mark m : MARKS.values()) if (m.data.owner().equals(me)) ids.add(m.data.id());
		for (UUID id : ids) requestRemove(id);
	}

	/** Deletes my mark closest to the crosshair (within ~6 degrees). */
	public static void removeLookedAt() {
		MinecraftClient mc = MinecraftClient.getInstance();
		if (mc.player == null) return;
		Vec3d eye = mc.player.getEyePos();
		Vec3d look = mc.player.getRotationVec(1.0f);
		UUID me = mc.player.getUuid();
		double best = Math.cos(Math.toRadians(6.0));
		UUID bestId = null;
		for (Mark m : MARKS.values()) {
			if (!m.data.owner().equals(me)) continue;
			Vec3d dir = new Vec3d(m.data.x() - eye.x, m.data.y() - eye.y, m.data.z() - eye.z);
			double len = dir.length();
			if (len < 1e-4) continue;
			double dot = dir.dotProduct(look) / len;
			if (dot > best) {
				best = dot;
				bestId = m.data.id();
			}
		}
		if (bestId != null) requestRemove(bestId);
	}

	private static void requestRemove(UUID id) {
		if (ClientPlayNetworking.canSend(RemovePayload.ID)) {
			ClientPlayNetworking.send(new RemovePayload(id));
		}
		// Local removal covers local-only marks (and is harmless if the server echoes it back).
		MARKS.remove(id);
	}

	public static String displayLabel(MarkData d) {
		if (d.label() != null && !d.label().isEmpty()) return d.label();
		return Text.translatable(MarkType.byOrdinal(d.type()).key()).getString();
	}
}

		settingsKey = KeyBindingHelper.registerKeyBinding(

		ClientPlayNetworking.registerGlobalReceiver(RemovePayload.ID,
				(payload, ctx) -> ClientMarks.remove(payload.id()));
		ClientPlayConnectionEvents.DISCONNECT.register((handler, client) -> ClientMarks.clear());

					client.setScreen(new WheelScreen(ClientMarks.aimPoint(client)));

				}
			}
		});

package com.markpoint.client;

import com.markpoint.MarkData;
import com.markpoint.MarkType;
import net.minecraft.client.MinecraftClient;
import net.minecraft.client.gui.DrawContext;
import net.minecraft.text.Text;
import net.minecraft.util.Util;
import net.minecraft.util.math.Vec3d;

public final class MarkRenderer {
	private static final String[] CARDINALS = {"S", "SW", "W", "NW", "N", "NE", "E", "SE"};
	private static final int MARGIN = 14;

	private MarkRenderer() {
	}

	public static void render(DrawContext ctx) {
		MinecraftClient mc = MinecraftClient.getInstance();
		if (mc.player == null || mc.options.hudHidden || ClientMarks.all().isEmpty()) return;
		ClientMarks.prune();

		Vec3d eye = mc.player.getEyePos();
		double yr = Math.toRadians(mc.player.getYaw());
		double pr = Math.toRadians(mc.player.getPitch());
		// camera basis (Minecraft yaw/pitch convention)
		double fx = -Math.sin(yr) * Math.cos(pr), fy = -Math.sin(pr), fz = Math.cos(yr) * Math.cos(pr);
		double rx = -Math.cos(yr), ry = 0, rz = -Math.sin(yr);
		double ux = ry * fz - rz * fy, uy = rz * fx - rx * fz, uz = rx * fy - ry * fx;

		int w = ctx.getScaledWindowWidth();
		int h = ctx.getScaledWindowHeight();
		double fov = mc.options.getFov().getValue();
		double focal = (h / 2.0) / Math.tan(Math.toRadians(fov) / 2.0);
		java.util.UUID me = mc.player.getUuid();
		long now = System.currentTimeMillis();
		double pulse = ClientConfig.animations ? 1.0 + 0.12 * Math.sin(Util.getMeasuringTimeMs() / 220.0) : 1.0;

		for (ClientMarks.Mark m : ClientMarks.all()) {
			MarkData d = m.data;
			MarkType type = MarkType.byOrdinal(d.type());
			if (!ClientConfig.enabled(type)) continue;

			double dx = d.x() - eye.x, dy = d.y() - eye.y, dz = d.z() - eye.z;
			double dist = Math.sqrt(dx * dx + dy * dy + dz * dz);
			if (dist > ClientConfig.maxDistance) continue;

			double zc = dx * fx + dy * fy + dz * fz;
			double xc = dx * rx + dy * ry + dz * rz;
			double yc = dx * ux + dy * uy + dz * uz;

			double sx = 0, sy = 0;
			boolean onScreen = false;
			if (zc > 0.1) {
				sx = w / 2.0 + xc / zc * focal;
				sy = h / 2.0 - yc / zc * focal;
				onScreen = sx >= MARGIN && sx <= w - MARGIN && sy >= MARGIN && sy <= h - MARGIN;
			}
			if (!onScreen) {
				if (!ClientConfig.showOffscreen) continue;
				double vx = xc, vy = -yc;
				double hx = w / 2.0 - MARGIN, hy = h / 2.0 - MARGIN;
				double len = Math.max(Math.abs(vx) / hx, Math.abs(vy) / hy);
				if (len < 1e-9) {
					vx = 0;
					vy = 1;
					len = 1.0 / hy;
				}
				sx = w / 2.0 + vx / len;
				sy = h / 2.0 + vy / len;
			}

			int color = (d.color() & 0x00FFFFFF) | 0xFF000000;
			int half = Math.max(2, (int) Math.round((onScreen ? 4.0 : 3.0) * ClientConfig.scale * pulse));
			int x = (int) Math.round(sx), y = (int) Math.round(sy);

			ctx.fill(x - half - 1, y - half - 1, x + half + 1, y + half + 1, 0xFF000000);
			ctx.fill(x - half, y - half, x + half, y + half, color);
			ctx.fill(x - 1, y + half + 1, x + 1, y + half + 5, 0xFF000000); // small pin
			ctx.fill(x, y + half + 1, x + 1, y + half + 4, color);

			if (!onScreen) continue;

			// name line (label + owner)
			String name = ClientMarks.displayLabel(d);
			if (ClientConfig.showOwner && !d.owner().equals(me) && !d.ownerName().isEmpty()) {
				name += " (" + d.ownerName() + ")";
			}
			ctx.drawCenteredTextWithShadow(mc.textRenderer, Text.literal(name), x, y - half - 12, color);

			// info line: direction, distance, remaining time
			StringBuilder info = new StringBuilder();
			info.append(cardinal(dx, dz));
			if (ClientConfig.showDistance) info.append("  ").append((int) dist).append("m");
			if (m.expiresAt > 0) {
				long sec = Math.max(0, (m.expiresAt - now + 999) / 1000);
				info.append("  ").append(sec >= 60 ? (sec / 60) + "m" + (sec % 60) + "s" : sec + "s");
			}
			ctx.drawCenteredTextWithShadow(mc.textRenderer, Text.literal(info.toString()),
					x, y + half + 8, 0xFFFFFFFF);
		}
	}

	private static String cardinal(double dx, double dz) {
		double yaw = Math.toDegrees(Math.atan2(-dx, dz));
		int idx = (int) Math.floor((((yaw + 22.5) % 360) + 360) % 360 / 45.0) % 8;
		return CARDINALS[idx];
	}

package com.markpoint.client;

import com.markpoint.MarkType;
import net.minecraft.client.gui.Click;
import net.minecraft.client.gui.DrawContext;
import net.minecraft.client.gui.screen.Screen;
import net.minecraft.text.Text;
import net.minecraft.util.Util;
import net.minecraft.util.math.Vec3d;

import java.util.List;

/** Radial wheel: pick a mark type with the mouse. Click the centre to delete your marks. */
public class WheelScreen extends Screen {
	private final Vec3d target;
	private final List<MarkType> types = ClientConfig.enabledTypes();
	private final long opened;

	public WheelScreen(Vec3d target) {
		super(Text.translatable("markpoint.wheel.title"));
		this.target = target;
		this.opened = Util.getMeasuringTimeMs();
	}

	@Override
	public boolean shouldPause() {
		return false;
	}

	private float progress() {
		if (!ClientConfig.animations) return 1f;
		float t = Math.min(1f, (Util.getMeasuringTimeMs() - opened) / 180f);
		return 1f - (1f - t) * (1f - t) * (1f - t); // ease-out
	}

	/** Index of hovered segment, -1 for centre / nothing. */
	private int hover(double mx, double my) {
		if (types.isEmpty()) return -1;
		double dx = mx - width / 2.0, dy = my - height / 2.0;
		if (Math.hypot(dx, dy) < 22) return -1;
		double a = Math.atan2(dy, dx) + Math.PI / 2;
		double full = Math.PI * 2;
		double seg = full / types.size();
		a = ((a % full) + full) % full;
		return (int) Math.round(a / seg) % types.size();
	}

	@Override
	public void render(DrawContext ctx, int mx, int my, float delta) {
		float p = progress();
		int alpha = (int) (0x70 * p);
		ctx.fill(0, 0, width, height, alpha << 24);

		int cx = width / 2, cy = height / 2;
		int n = types.size();
		double radius = 70 * p;
		int hov = hover(mx, my);
		boolean centerHover = Math.hypot(mx - cx, my - cy) < 22;

		// centre button (delete my marks)
		int cbox = 14;
		ctx.fill(cx - cbox - 1, cy - cbox - 1, cx + cbox + 1, cy + cbox + 1, centerHover ? 0xFFFFFFFF : 0xFF000000);
		ctx.fill(cx - cbox, cy - cbox, cx + cbox, cy + cbox, centerHover ? 0xFFB71C1C : 0xFF3A3A3A);
		ctx.drawCenteredTextWithShadow(textRenderer, Text.literal("X"), cx, cy - 4, 0xFFFFFFFF);

		for (int i = 0; i < n; i++) {
			MarkType t = types.get(i);
			double ang = -Math.PI / 2 + i * (Math.PI * 2 / n);
			int ix = cx + (int) Math.round(Math.cos(ang) * radius);
			int iy = cy + (int) Math.round(Math.sin(ang) * radius);
			boolean hv = i == hov;
			int half = hv ? 20 : 17;
			int color = ClientConfig.colors[t.ordinal()] | 0xFF000000;
			ctx.fill(ix - half - 2, iy - half - 2, ix + half + 2, iy + half + 2, hv ? 0xFFFFFFFF : 0xFF000000);
			ctx.fill(ix - half, iy - half, ix + half, iy + half, color);
			if (t == ClientConfig.favoriteType()) {
				ctx.drawCenteredTextWithShadow(textRenderer, Text.literal("*"), ix, iy - half + 3, 0xFF000000);
			}
			if (p > 0.6f) {
				ctx.drawCenteredTextWithShadow(textRenderer, Text.translatable(t.key()), ix, iy + half + 5,
						hv ? 0xFFFFFFFF : 0xFFBBBBBB);
			}
		}
	}

	@Override
	public boolean mouseClicked(Click click, boolean doubled) {
		if (click.button() == 0) {
			double mx = click.x(), my = click.y();
			int h = hover(mx, my);
			if (h >= 0) {
				ClientMarks.create(types.get(h), target);
				close();
				return true;
			}
			if (Math.hypot(mx - width / 2.0, my - height / 2.0) < 22) {
				ClientMarks.clearMine();
				close();
				return true;
			}
		}
		close();
		return true;
	}
}
